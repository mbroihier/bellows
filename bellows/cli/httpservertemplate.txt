def sigint_handler(_, __):
    background_object = HTTPServerBackground()
    LOGGER.warning("\nshutting down server....")
    background_object.continue_loop = False

class HTTPHandler(httpServer.BaseHTTPRequestHandler):
    '''
    HTTPHandler that handles GET requests of HTTP protocol - POSTs are ignored
    '''
    def do_GET(self):
        '''
        HTTP GET processing
        '''
        print("got a get with:", self.path)
        if self.path == '/':
            self.send_response(301)
            self.send_header('Location', '/index.html')
            self.end_headers()
        elif self.path == '/index.html':
            with open('./index.html', 'r') as file_object:
                content = ""
                for line in file_object:
                    content += line
                content = content.encode('utf-8')
                self.send_response(200)
                self.send_header('Content-Type', 'text')
                self.send_header('Content-Length', len(content))
                self.end_headers()
                self.wfile.write(content)
        elif self.path == '/lt.html':
            with open('./lt.html', 'r') as file_object:
                content = ""
                for line in file_object:
                    content += line
                content = content.encode('utf-8')
                self.send_response(200)
                self.send_header('Content-Type', 'text')
                self.send_header('Content-Length', len(content))
                self.end_headers()
                self.wfile.write(content)
        elif (self.path[-3:] == 'on?' or self.path[-4:] == 'off?' or self.path[-2:] == 'on'
              or self.path[-3:] == 'off'):
            command = self.path.replace('/', '').replace('?', '')
            try:
                with connect("ws://localhost:8126") as conn:
                    conn.send("http server internal connection request")
                    conn.socket.settimeout(5.0)
                    message = conn.recv()
                    LOGGER.info(f"received initial/connection status of: {message}")
                    device = command.split('o')[0]
                    LOGGER.info(f"device: {device}, command: {command}")
                    if json.loads(message)[device] in command:
                        LOGGER.warning(f"{json.loads(message)[device]} matches commanded state")
                    else:
                        conn.send(command)
                        message = conn.recv()
                        LOGGER.info(f"received final status of: {message}")
            except socket.timeout:
                LOGGER.error("TimeoutError while receiving status, terminating")
            except Exception as e:
                LOGGER.error(f"Unexpected Exception detected {e}")
            self.send_response(302)
            self.send_header('Location', '/index.html')
            self.end_headers()
        elif 'setCT' in self.path:
            command = self.path.replace('/', '').replace('?', '').replace('%20', ' ')
            try:
                with connect("ws://localhost:8126") as conn:
                    conn.send("http server internal connection request")
                    conn.socket.settimeout(5.0)
                    message = conn.recv()
                    LOGGER.info(f"received initial/connection status of: {message}")
                    LOGGER.info(f"command: {command}")
                    conn.send(command)
                    message = conn.recv()
                    LOGGER.info(f"received final status of: {message}")
            except socket.timeout:
                LOGGER.error("TimeoutError while receiving status, terminating")
            except Exception as e:
                LOGGER.error(f"Unexpected Exception detected {e}")
            self.send_response(302)
            self.send_header('Location', '/lt.html')
            self.end_headers()
        else:
            self.send_error(404)
            self.end_headers()

    def do_POST(self):
        '''
        HTTP POST processing
        '''
        length = int(self.headers['Content-Length'])
        line = self.rfile.read(length)
        print("got a post with:", line)

class HTTPServerBackground():
    '''
    HTTP Server Background Thread - as a singleton
    '''
    def __new__(cls):
        '''
        Thread object allocation
        '''
        if not hasattr(cls, 'instance'):
            cls.instance = super(HTTPServerBackground, cls).__new__(cls)
        return cls.instance

    def __init__(self):
        '''
        Constructor - all attributes are truely defined outside of __init__
        '''
        if not hasattr(self, 'terminate'):
            self.terminate = None
            self.server = None
            self.continue_loop = None

    def background(self):
        '''
        Request message handling in background
        '''
        while not self.terminate:
            self.server.handle_request()

    def shutdown(self):
        '''
        Shutdown/terminate the server thread
        '''
        self.terminate = True

@click.command()
@click_log.simple_verbosity_option(logging.getLogger(), default="INFO")
@util.background
async def main():
    '''
    Test main program for server
    '''
    global continueLoop
    logging.basicConfig(format='%(asctime)s %(levelname)-8s [%(filename)s:%(lineno)4d] %(message)s',
                        datefmt=' %Y-%m-%d:%H:%M:%S', level=LOGGER.getEffectiveLevel())
    signal.signal(signal.SIGINT, sigint_handler)

    address = ('', 8124)
    server = httpServer.HTTPServer(address, HTTPHandler)
    ***begin insert***
    background_object = HTTPServerBackground()
    background_object.terminate = False
    background_object.server = server
    background_object.continue_loop = True

    background_thread = threading.Thread(target=background_object.background)
    try:
        background_thread.start()
    except Exception as e:
        LOGGER.warning(f"Received a {e} exception")
        background_object.continue_loop = False
    while background_object.continue_loop:
        try:
            time.sleep(1.0)
        except Exception:
            break
    background_object.shutdown()
    response = requests.get('http://localhost:8124/index.html', timeout=5.0)
    LOGGER.debug(f"got {response}")
    background_thread.join()
if __name__ == "__main__":
    asyncio.run(main())
